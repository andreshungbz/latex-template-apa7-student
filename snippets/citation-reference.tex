\subsection{Citation \& Reference Usage}
\begin{itemize}
    \item Parenthetical: This is a cited statement \parencite{Lu2023}.
    \item Narrative: \textcite{Lu2023} did research.
    \item Multiple: \parencite{Lu2023, Seiti2023} corroborate the findings.
    \item Figure: Refer to Figure~\ref{fig:figure}.
\end{itemize}
